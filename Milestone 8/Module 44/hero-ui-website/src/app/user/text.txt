import UserCardInfo from '../components/UserCard';
import fs from "fs";
import path from "path";

const filePath = path.join(process.cwd(), "src/data/contacts.json");


const AboutUs = async() => {
    const data = fs.readFileSync(filePath, "utf-8");
    const users = JSON.parse(data);
    return (
      <>
        
        <div className="flex flex-col items-center justify-center py-2 space-y-5">
          {users.map((user) => (
            <UserCardInfo key={user.id} user={user}></UserCardInfo>
          ))}
        </div>
      </>
    );
};

export default AboutUs;
